; x86-64, Intel syntax
; rdi 指向输出缓冲区 buf
; 虽然 ai 读汇编很快，但希望你尽量用肉眼和手敲推出来，看不懂的指令先查再做

start:
    lea     rdi, [buf]              ; rdi = buf 的地址（输出目标）

    lea     rsi, [byte_404000]      ; rsi = 数据源1 的地址
    mov     ecx, 19                 ; ecx = 19，循环 19 次

loc_401000:
    mov     al, byte ptr [rsi]      ; al = *rsi（取一个字节）
    mov     byte ptr [rdi], al      ; *rdi = al（写入输出）
    inc     rsi                     ; rsi++（源指针后移）
    inc     rdi                     ; rdi++（目标指针后移）
    loop    loc_401000              ; ecx--；若 ecx != 0 则跳回 loc_401000

    ; —— 下面是一段"恒为真"的判断，用来决定走哪个分支 ——
    mov     eax, 0x2a               ; eax = 0x2a = 42
    add     eax, 0x16               ; eax = 42 + 22 = 64 = 0x40
    cmp     eax, 0x40               ; 比较 eax 与 0x40
    jne     loc_401080              ; 不相等就跳（实际相等，不跳）

    mov     ebx, 0x10               ; ebx = 0x10 = 16
    shl     ebx, 2                  ; ebx <<= 2，即 ebx = 16 * 4 = 64 = 0x40
    cmp     eax, ebx                ; 比较 eax(0x40) 与 ebx(0x40)
    jne     loc_401080              ; 不相等就跳（实际相等，不跳）

    mov     ecx, 0x39               ; ecx = 0x39 = 57
    sub     ecx, 0x20               ; ecx = 57 - 32 = 25 = 0x19
    cmp     ecx, 0x18               ; 比较 ecx(25) 与 0x18(24)
    jg      loc_401050              ; 若 ecx > 0x18 则跳（25 > 24，成立，跳！）

    jmp     loc_401080              ; 否则跳到这里（本次不执行）

    ; —— 真正走到的分支：loc_401050 ——
loc_401050:
    lea     rsi, [byte_404020]      ; rsi = 数据源2 的地址
    mov     ecx, 31                 ; ecx = 31，循环 31 次

loc_401060:
    mov     al, byte ptr [rsi]      ; al = *rsi
    xor     al, 0x42                ; al ^= 0x42（与 0x42 异或）
    mov     byte ptr [rdi], al      ; *rdi = al
    inc     rsi                     ; rsi++
    inc     rdi                     ; rdi++
    loop    loc_401060              ; ecx--；若 ecx != 0 跳回

    jmp     loc_4010b0              ; 跳到收尾

    ; —— 未走到的分支：loc_401080 ——
loc_401080:
    lea     rsi, [byte_404030]      ; rsi = 数据源3 的地址
    mov     ecx, 9                  ; ecx = 9，循环 9 次

loc_401090:
    mov     al, byte ptr [rsi]      ; al = *rsi
    mov     byte ptr [rdi], al      ; *rdi = al
    inc     rsi
    inc     rdi
    loop    loc_401090

loc_4010b0:
    mov     byte ptr [rdi], 0       ; 在输出末尾写入 '\0'（字符串结束符）
    ret

; —— 数据段 ——
byte_404000:
    db 0x66, 0x6c, 0x61, 0x67, 0x7b, 0x55, 0x6e, 0x64, 0x65, 0x72
    db 0x73, 0x74, 0x61, 0x6e, 0x64, 0x69, 0x6e, 0x67, 0x5f
    ; 共 19 字节

byte_404020:
    db 0x23, 0x31, 0x31, 0x27, 0x2f, 0x20, 0x2e, 0x3b, 0x1d, 0x2e
    db 0x23, 0x2c, 0x25, 0x37, 0x23, 0x25, 0x27, 0x1d, 0x2b, 0x31
    db 0x1d, 0x2b, 0x2f, 0x32, 0x2d, 0x30, 0x36, 0x23, 0x2c, 0x36
    db 0x3f
    ; 共 31 字节

byte_404030:
    db 0x63, 0x6f, 0x72, 0x72, 0x65, 0x63, 0x74, 0x21, 0x7d
    ; 共 9 字节

buf:
    db 64 dup(0)                    ; 64 字节输出缓冲区，全部初始化为 0
